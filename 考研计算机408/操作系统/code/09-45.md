semaphore buff = 0; // 表示N个单元的缓冲区
semaphore mutex = 1;  // 互斥访问该缓冲区
P1() {
    int x = produce();
    P(mutex);
    put(x);
    V(buff);
    V(mutex);
}

P2() {
    P(mutex);
    P(buff);
    getodd();  // 取出一个奇数
    countodd();  // 统计奇数个数
    V(mutex);
}

P3() {
    P(mutex);
    P(buff);

    V(mutex);
