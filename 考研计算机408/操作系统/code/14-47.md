```c
sema buff = 1000; // 表示当前缓冲区的空闲数量
sema mutex = 1;  // 用于互斥访问缓冲区
sema consumer = 1; // 用于表示消费者当前是否能取出产品
producers() {
    while (1) {
        P(buff);
        P(mutex);
        生产产品放入缓冲区;
        V(mutex);
    }
}

consumers() {
    while (1) {
        // 一个消费者连续取出10件产品
        P(consumer);
        for (int i = 0; i < 10; i++) {
            P(mutex);
            从缓冲区取出1件产品;
            V(buff);
            V(mutex);
        }
        V(consumer);
    }
}
