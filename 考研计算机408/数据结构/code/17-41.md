void inOrder(BTree *root, int depth) {
    // 先对该顶点做一些基本的判断
    if (!root) return;
    // 如果是叶子结点则表示是操作数，直接输出即可
    if (!root->left && !root->right) {
        printf("%s", root->data);
        return;
    }

    // 进行递归遍历前先输出括号包住，非根才输出括号
    if (depth>1)
        printf("(");
    //左
    inOrder(root->left, depth+1);

    //中————输出运算符
    printf("%s", root->data);

    inOrder(root->right, depth+1);

    if (depth>1)
        printf(")");
}

void express(BTree *root) {
    inOrder(root, 1);
}
