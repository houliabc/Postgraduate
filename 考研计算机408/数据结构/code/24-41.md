（1）初始化一个入度表，用于存放每个顶点的入度情况，初始时都为0，接着遍历邻接矩阵，统计每个顶点的入度个数；随后初始化一个队列，将入度为0的顶点入队，检查队列长度，如果任何时刻队列中出现了2个及以上的顶点个数，则该拓扑序列肯定是不唯一的，接着只要队列不为空就一直重复如下的操作：
出队一个顶点，并将所有其他入度>0的顶点，若存在起始顶点是该出队顶点，终点顶点是该入度>0的顶点，则将入度-1，如果此刻入度为0，则加入队列，并实时检查队列长度是否>1，一旦＞1就返回0
如果队列遍历完后，拓扑序列个数和顶点个数相同，则返回1，否则说明没有拓扑序列，返回0

typedef struct {
    int numVertices, numEdges;   // 图的顶点数和有向边数
    char VerticesList[MAXV];   // 顶点表
    int Edge[MAXV][MAXV];  // 邻接矩阵
} MGraph;
int uniquely(MGraph G) {
    int n = G.numVertices, inDegree[n];
    for (int i = 0; i < n; i++) inDegree[i]=0;
    // 遍历邻接矩阵
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (Edge[i][j] == 1)
                inDegree[j]++;
    // 初始化一个n大小的队列
    Queue que;
    InitQueue(que, n);
    // 将度为0的顶点入队
    for (int i = 0; i < n; i++) {
        if (inDegree[i] == 0) {
            enQueue(VerticesList[i]);
            // 检查队列长度，超过1则拓扑序列一定不唯一
            if (que.size > 1)
                return 0;
        }
    }

    // 记录已输出顶点个数
    int count = 0;
    while (!IsQueueEmpty(que)) {
        // 获取该顶点元素
        char c = getQueueFront();
        // 查询该元素下标
        int i = getIndex(G, c);
        deQueue(que);
        count++;
        // 将以该顶点为起始点的其他顶点的入度-1
        for (int j = 0; j < n; j++) {
            if (inDegree[j] > 0 && Edge[i][j] == 1) {
                inDegree[j]--;
                if (inDegree[j] == 0) {
                    enQueue(VerticesList[j]);
                    // 检查队列长度，超过1则拓扑序列一定不唯一
                    if (que.size > 1)
                        return 0;
                }
            }
        }
    }

    // 遍历完后若输出顶点个数和顶点个数相同，则有唯一拓扑序列
    if (count < n)
        return 0;
    return 1;
}
