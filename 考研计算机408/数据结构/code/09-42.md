（3）
```c
int findK(LNode *list, int k) {
    // 获取链表长度
    int n = list->length;
    LNode *slow = list, *fast = list;
    // 判断k是否合规且先让fast向后移动k-1个位置
    if (k <= 0 || k > n || list->link == NULL)
        return 0;
    for (int i = 0; i < k - 1; i++) {
        // 前面已校验过确保不会出现访问空结点的link
        fast = fast->link;
    }
    // 一直到fast指向尾节点
    while (fast->link) {
        fast = fast->link;
        slow = slow->link;
    }
    // 此时slow指向题目要求的结点
    // 输出
    printf("%d", slow->data);
    return 1;
}
时间复杂度O（n），空间复杂度O（1）
