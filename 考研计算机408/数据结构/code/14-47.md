```c
semaphore mutex=1; // 互斥访问缓存区
semaphore empty=1000;  // 该缓存区还能存放多少件产品
semaphore full=0;  // 该缓存区已有多少件产品
生产者i() {
    while () {
        生产产品();
        P(empty);
        P(mutex);
        放入产品();
        V(mutex);
        V(full);
    }
}
消费者i() {
    while () {
        // 如果缓存区有产品，则该消费者连续拿取10件产品
        for(int i=0; i<10;i++) {
            // 只要有产品就拿取
            P(full);
            P(mutex);
            取出产品();
            V(mutex);
            V(empty);
        }
    }
}
