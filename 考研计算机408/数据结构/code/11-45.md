```c
cobegin
semaphore seats = 10;  // 表示可以供给顾客等待的座位的数量
semaphore mutex = 1;  // 用于互斥访问取号机
semaphore wait = 0;  // 用于顾客通知营业员获取服务
process 顾客 i {
    P(seats);
    P(mutex);
    从取号机获得一个号码;
    V(mutex);
    等待叫号;
    V(wait);
    获得服务;
    V(seats);
}
process 营业员 {
    while (TRUE) {
        P(wait);
        叫号;
        为顾客服务;
    }
}
