（1）首先遍历一遍链表统计表长n，根据表长n为奇数则取中点位序m为n/2+1，而如果是偶数则取m=n/2。接着将m作为头结点，依次遍历后面的元素，将后面的元素依次头插到头结点后面。最后，记录位序m结点，一个指针指向第一个结点，一个指向位序为m+1的结点，重复执行如下操作：每次将第二个指针的结点插入到第一个指针后面，将第一个指针后移两位，第二个指针后移一位，当第一个指针指向位序m结点时，排列完毕
（2）
```c
typedef struct node{
    int data;
    struct node *next;
}NODE;
void rearrange(NODE *head) {
    if (!head || !head->next)
        return;
    NODE *p = head->next, q, mid, t;
    int n = 0, m;
    while (p) {
        n++;
        p = p->next;
    }
    if (n % 2 == 1) m = n / 2 + 1;
    else m = n/2;
    // 找到位序m的结点
    p = head->next;
    for (int i = 1; p; p=p->next, i++) {
        if (i == m) break;
    }
    mid = m; p = mid->next, mid->next = NULL;  // 第一个头插后变成尾节点
    // 以mid为头结点，将后续结点头插到mid后
    while (p) {
        p->next = mid->next;
        mid->next = p;
        p = p->next;
    }
    p = head->next, q = mid->next;
    while (p != m) {
        t = q->next;
        q->next = p->next;
        p->next = q;
        // 下一轮
        p = q->next;
        q = t;
    }
}
