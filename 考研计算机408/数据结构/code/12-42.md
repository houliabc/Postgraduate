（1）若两个单词有相同的后缀，那么这两单词的最后n位是相同的。那么首先遍历一下两个链表的长度n1和n2，将指针指向长度最长的链表从头结点开始向后移动|n1-n2|的位置，之后两个指针同时后移，当遇到的字符相同时，说明找到了相同的后缀，返回该结点。
（2）
```c
typedef struct node {
    char data;
    struct node *next;
} Node;
Node *getSame(Node *str1, Node *str2) {
    if (!str1 || !str2) return NULL;
    Node *p1 = str1->next, *p2 = str2->next;
    int n1 = 0, n2 = 0;
    while (p1) {p1=p1->next; n1++;}
    while (p2) {p2=p2->next; n2++;}
    p1 = str1->next; p2 = str2->next
    // 让p1指向长的链表
    if (n2 > n1) {
        int t = n1; n1 = n2; n2 = t;
        p1 = str2->next;
        p2 = str1->next;
    }
    // 让长的链表往后移动n1-n2个位置
    int i = 0;
    while (p1 && i < n1-n2) {
        p1 = p1->next;
        i++;
    }
    // 此时指针都位于两个链表的倒数第相同的位置上，同时后移比较
    while (p1 && p2) {
        if (p1->data == p2->data) return p1;
        p1 = p1->next;
        p2 = p2->next;
    }
    // 表示没有找到相同后缀
    return NULL;
}
