LNode *getHouzhui(LNode *str1, LNode *str2) {
    if (!str1 || !str2)
        return NULL;
    LNode *p1 = str1->next, *p2 = str2->next, *t, *pre = NULL;
    int k = 1;  // 设置反转前的第一个结点的指针为空
    // 实现两个链表的反转
    while (p1) {
        t = p1->next;
        p1->next = str1->next;
        if (k-- == 1)
            p1->next = NULL;
        str1->next = p1;
        p1 = t;
    }
    k = 1;
    while (p2) {
        t = p2->next;
        p2->next = str1->next;
        if (k-- == 1)
            p1->next = NULL;
        str1->next = p2;
        p2 = t;
    }
    // 接着找到后缀不相同的前一个位置
    p1 = str1->next;
    p2 = str2->next;
    while (p1 && p2) {
        if (p1->data == p2->data) {
            p1 = p1->next;
            p2 = p2->next;
        }
        // 返回当前不同的前一个位置
        else 
            return pre;
        pre = p1;
    }
    // 两个链表同时为空，表示两个单词全部相同，返回头指针；否则返回pre
    if (!p1 && !p2)
        return str1;
    else
        return NULL;
}
