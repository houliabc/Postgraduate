int abs(int n) {
    return n>=0 ? n : -n;
}
Node *unique(Node *HEAD) {
    if (!HEAD || !HEAD->link)
        return NULL;
    Node *pre = HEAD, *p = HEAD->link, *t;
    HashTable h;
    InitHashTable(&h, 1000);  // 根据实际情况修改大小
    while (p) {
        // 查找当前元素绝对值是否在哈希表中，在就删除，不在就插入哈希表
        if (getHashTable(h, abs(p->data))) {
            t = p->link;
            pre->link = t;
            free(p);
            p = t;
        }
        else {
            putHashTable(&h, abs(p->data));
        }
        pre = p;
        p = p->link;
    }
    return HEAD;
}
